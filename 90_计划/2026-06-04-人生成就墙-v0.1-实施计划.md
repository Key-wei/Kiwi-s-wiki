# 人生成就墙 v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 7 个月内 solo 副业(每周 10-20 小时)完成「人生成就墙」微信小程序 v0.1——一个让中国白领能像看 Steam 个人主页一样查看人生进度的产品,核心功能为 200 个预设成就的图鉴 + 手动勾选 + **今日 5 个推送 (A)** + **10 个 Persona 身份徽章 (B)** + 分享卡片。

**Architecture:** 微信小程序原生(WXML/JS,无框架)+ 微信云开发(云函数 + 云数据库 + 云存储)。前后端零运维,客户端用 Canvas 端侧生成分享卡片。新用户进入自动点亮主线 8 个成就(新手大礼包)+ 3-4 步气泡引导;首页采用「今日 5 个推送」(A 模块)解决决策疲劳;长期目标感由 Persona 身份徽章系统(B 模块)驱动。

**Tech Stack:**
- 微信小程序原生 (WXML / WXSS / JS)
- 微信云开发 (云函数 / 云数据库 MongoDB / 云存储)
- Jest (云函数 + utils 单元测试)
- Canvas API (分享卡片生成)
- Git + GitHub (版本管理)
- 无前端框架 (不用 Taro / uni-app,降低复杂度)

```javascript
// 等级阈值(从 Lv 1 开始)
// 解锁数 → 等级映射:0→1, 5→2, 15→3, 30→4, 50→5, 75→6, 105→7, 140→8, 175→9, 200→10
const LEVEL_THRESHOLDS = [0, 5, 15, 30, 50, 75, 105, 140, 175, 200];

function calculateLevel(unlockedCount) {
  if (unlockedCount < 0) {
    throw new Error('unlockedCount 不能为负数');
  }
  let level = 1;
  for (let i = 0; i < LEVEL_THRESHOLDS.length; i++) {
    if (unlockedCount >= LEVEL_THRESHOLDS[i]) {
      level = i + 1;
    }
  }
  return level;
}

module.exports = { calculateLevel, LEVEL_THRESHOLDS };
```

```javascript
function rarityFromRate(rate) {
  if (rate === -1) return 'hidden';
  if (rate >= 50) return 'common';
  if (rate >= 10) return 'elite';
  if (rate >= 1) return 'epic';
  return 'legendary';
}

const COLOR_MAP = {
  common: '#999999',
  elite: '#4A90E2',
  epic: '#9B59B6',
  legendary: '#F39C12',
  hidden: '#2C3E50',

function rarityColor(rarity) {
  return COLOR_MAP[rarity] || COLOR_MAP.common;
```

```javascript
// 推荐算法核心
const cloud = require('wx-server-sdk');
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV });
const db = cloud.database();
const _ = db.command;

exports.main = async (event) => {
  const { OPENID } = cloud.getWXContext();
  const { theme } = event; // category name from frontend

  // 1. 取该类别下用户未勾选的成就
  const unlockedRes = await db.collection('userAchievements')
    .where({ openid: OPENID })
    .field({ achievementId: true })
    .get();
  const unlockedIds = unlockedRes.data.map(d => d.achievementId);

  // 2. 取用户最近 7 天跳过的成就(冷却)
  const sevenDaysAgo = new Date(Date.now() - 7 * 24 * 60 * 60 * 1000);
  const skippedRes = await db.collection('dailyPush')
    .where({
      openid: OPENID,
      date: _.gte(sevenDaysAgo),
    })
    .get();
  const skippedIds = new Set();
  skippedRes.data.forEach(p => {
    Object.entries(p.actions || {}).forEach(([id, a]) => {
      if (a === 'skipped') skippedIds.add(parseInt(id));
    });
  });

  // 3. 候选池: 该类别 - 已解锁 - 7天内跳过
  const candidatesRes = await db.collection('achievements')
    .where({
      category: theme,
      _id: _.nin(unlockedIds.map(id => `ach_${id}`)),
    })
    .get();
  const candidates = candidatesRes.data.filter(a => !skippedIds.has(a.id));

  // 4. 按稀有度分层抽样: 50% 普通 + 30% 精英 + 20% 史诗+
  const byRarity = {
    common: candidates.filter(a => a.rarity === 'common'),
    elite:  candidates.filter(a => a.rarity === 'elite'),
    high:   candidates.filter(a => ['epic','legendary'].includes(a.rarity)),
  };

  const picks = [
    ...sample(byRarity.common, 2),
    ...sample(byRarity.elite, 2),
    ...sample(byRarity.high, 1),
  ].slice(0, 5);

  // 5. 持久化今日推送记录
  const today = new Date().toISOString().slice(0, 10);
  await db.collection('dailyPush').add({
    data: {
      openid: OPENID,
      date: today,
      theme,
      achievementIds: picks.map(p => p.id),
      actions: {},
    }
  });

  return { picks };
};

function sample(arr, n) {
  return arr.sort(() => Math.random() - 0.5).slice(0, n);
}
```
